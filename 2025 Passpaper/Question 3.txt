CREATE SCHEMA 2025_passpaper;

USE 2025_passpaper;

create table student
(
studentID int auto_increment not null,
student_name varchar(20) not null,
student_surname varchar(20) not null,
student_number varchar(6) not null,
primary key (studentID)
);
desc student;

create table lecturer
(
lecturerID int auto_increment not null,
lecturer_name varchar(20) not null,
lecturer_surname varchar(20) not null,
primary key (lecturerID)
);


create table tutorial
(
TutorialID int auto_increment not null,
studentID int not null,
lecturerID int not null,
tutorialDate Date not null,
tutorialDate Time not null,
TutorialDuration int not null,
primary key (TutorialID),
foreign key (studentID) references student(studentID),
foreign key (lecturerID) references lecturer(lecturerID)
);

desc tutorial;
